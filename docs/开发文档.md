  🔴 Critical（必须解决）

  1. 并行执行缺失（性能瓶颈）

  问题：所有查询串行执行，延迟高

  现状代码（components/diagnose/handlers.py 推测）：
  async def handle_diagnose_query(state: TCMAgentState):
      # 串行执行
      kg_result = await query_kg(state.symptoms)
      case_result = await search_cases(state.symptoms)
      classic_result = await search_classics(state.symptoms)

      # 综合分析
      answer = await synthesize(kg_result, case_result, classic_result)
      return {"answer": answer}

  改进方案 1：使用 asyncio.gather（最小改动）
  async def handle_diagnose_query(state: TCMAgentState):
      # 并行执行
      kg_task = query_kg(state.symptoms)
      case_task = search_cases(state.symptoms)
      classic_task = search_classics(state.symptoms)

      kg_result, case_result, classic_result = await asyncio.gather(
          kg_task, case_task, classic_task
      )

      # 综合分析
      answer = await synthesize(kg_result, case_result, classic_result)
      return {"answer": answer}

  改进方案 2：使用 LangGraph Send()（推荐）
  # 创建专门的工具执行节点
  async def execute_tool_node(state: ToolTaskState):
      """执行单个工具任务"""
      tool_type = state["tool_type"]
      query = state["query"]

      if tool_type == "kg":
          result = await query_kg(query)
      elif tool_type == "case":
          result = await search_cases(query)
      elif tool_type == "classic":
          result = await search_classics(query)

      return {"result": result}

  # 在主图中添加并行分发
  def map_tasks(state: TCMAgentState):
      """并行分发任务"""
      return [
          Send("execute_tool", {"tool_type": "kg", "query": state.symptoms}),
          Send("execute_tool", {"tool_type": "case", "query": state.symptoms}),
          Send("execute_tool", {"tool_type": "classic", "query": state.symptoms}),
      ]

  # 在图构建时
  graph.add_node("execute_tool", execute_tool_node)
  graph.add_conditional_edges("diagnose_planner", map_tasks, ["execute_tool"])
  graph.add_edge("execute_tool", "diagnose_synthesizer")  # 自动 Reduce

  优先级：⭐⭐⭐⭐⭐（最高）
  预期收益：延迟降低 50-70%

  2. 大结果未自动驱逐（上下文溢出风险）

  问题：Neo4j 查询可能返回大量数据，直接放入上下文会导致 Token 溢出

  现状：
  # 当前实现（推测）
  kg_result = await query_kg(symptoms)  # 可能返回 50k tokens
  state.cypher_queries.append(kg_result)  # 直接存入状态

  改进方案：集成 FilesystemMiddleware
  from deepagents.middleware.filesystem import FilesystemMiddleware
  from deepagents.backends import CompositeBackend, StateBackend, StoreBackend

  # 在中间件链中添加
  filesystem_middleware = FilesystemMiddleware(
      backend=lambda rt: CompositeBackend(
          default=StateBackend(rt),
          routes={
              "/kg_results/": StateBackend(rt),      # 临时存储
              "/case_library/": StoreBackend(rt),    # 持久化
              "/classics/": StoreBackend(rt),        # 持久化
          }
      ),
      tool_token_limit_before_evict=15000,  # 超过 15k tokens 自动写入文件
      system_prompt="""
  当工具返回大量结果时：
  1. 使用 write_file 保存完整结果
  2. 上下文中只保留摘要（前 200 tokens）
  3. 需要详细内容时使用 read_file 读取
  """
  )

  middleware_chain.add(filesystem_middleware, priority=3)

  优先级：⭐⭐⭐⭐⭐（最高）
  预期收益：防止上下文溢出，支持超大查询结果

  from fastapi import APIRouter
  from langgraph.types import Command

  router = APIRouter()

  @router.post("/diagnosis/approve")
  async def approve_diagnosis(thread_id: str, decision: str):
      """医生审批接口"""
      config = {"configurable": {"thread_id": thread_id}}

      if decision == "approve":
          result = await tcm_graph.ainvoke(
              Command(resume={"decisions": [{"type": "approve"}]}),
              config=config
          )
      elif decision == "reject":
          result = await tcm_graph.ainvoke(
              Command(resume={"decisions": [{"type": "reject", "message": "..."}]}),
              config=config
          )

      return {"status": "success", "result": result}

  优先级：⭐⭐⭐⭐⭐（医疗合规必需）
  预期收益：满足医疗合规要求
