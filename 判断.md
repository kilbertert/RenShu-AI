# backend/scripts/explore_tcm_dataset.py
"""
探查 TCM_Dataset 目录结构与文件内容

执行：python -m scripts.explore_tcm_dataset
输出：探索报告到 backend/docs/tcm_dataset_exploration_report.md
"""
import os
import csv
from pathlib import Path
from collections import defaultdict

TCM_ROOT = Path("D:/AI/project/RenShu-AI/TCM_Dataset")
REPORT = Path(__file__).parent.parent / "docs" / "tcm_dataset_exploration_report.md"

def detect_encoding(file_path: Path) -> tuple[str, str]:
    """检测文件编码（GBK / UTF-8 兜底）"""
    for enc in ("utf-8", "gbk", "utf-8-sig"):
        try:
            with open(file_path, encoding=enc) as f:
                f.read(1024)
            return enc, "ok"
        except (UnicodeDecodeError, LookupError):
            continue
    return "unknown", "failed"

def detect_delimiter(file_path: Path, encoding: str) -> str:
    """通过首行判断分隔符"""
    with open(file_path, encoding=encoding) as f:
        first_line = f.readline()
    if "\t" in first_line:
        return "\\t"
    if "," in first_line:
        return ","
    if ";" in first_line:
        return ";"
    return "<unknown>"

def explore_file(file_path: Path) -> dict:
    """探查单个数据文件"""
    encoding, status = detect_encoding(file_path)
    if status == "failed":
        return {"path": str(file_path), "error": "无法识别编码"}

    delimiter = detect_delimiter(file_path, encoding)
    with open(file_path, encoding=encoding) as f:
        reader = csv.reader(f, delimiter="\t" if delimiter == "\\t" else delimiter)
        rows = list(reader)

    if not rows:
        return {"path": str(file_path), "error": "空文件"}

    return {
        "path": str(file_path),
        "encoding": encoding,
        "delimiter": delimiter,
        "row_count": len(rows) - 1,  # 减去表头
        "field_names": rows[0],
        "sample_row": rows[1] if len(rows) > 1 else None,
    }

def main():
    report_lines = ["# TCM_Dataset 探查报告\n\n"]
    field_mapping_lines = ["# 字段映射表\n\n"]

    for db_dir in sorted(TCM_ROOT.iterdir()):
        if not db_dir.is_dir() or db_dir.name == "all_mol2":
            continue
        report_lines.append(f"## {db_dir.name}\n\n")

        for sub_dir in sorted(db_dir.iterdir()):
            if not sub_dir.is_dir():
                continue
            report_lines.append(f"### {sub_dir.name}\n\n")
            report_lines.append("| 文件 | 编码 | 分隔符 | 行数 | 字段 | 样例 |\n")
            report_lines.append("|------|------|--------|------|------|------|\n")

            for f in sorted(sub_dir.iterdir()):
                if not f.is_file() or f.suffix not in (".csv", ".tsv", ".txt"):
                    continue
                info = explore_file(f)
                if "error" in info:
                    report_lines.append(f"| {f.name} | - | - | - | - | ❌ {info['error']} |\n")
                    continue

                sample = str(info["sample_row"])[:80] if info["sample_row"] else "-"
                fields = ", ".join(info["field_names"][:5])
                if len(info["field_names"]) > 5:
                    fields += f" ... (+{len(info['field_names'])-5})"

                report_lines.append(
                    f"| {f.name} | {info['encoding']} | {info['delimiter']} | "
                    f"{info['row_count']} | {fields} | {sample} |\n"
                )

                # 字段映射（粗略）
                field_mapping_lines.append(
                    f"## {db_dir.name} / {sub_dir.name} / {f.name}\n\n"
                    f"- 字段：`{', '.join(info['field_names'])}`\n"
                    f"- 行数：{info['row_count']}\n"
                    f"- **待 P0 阶段确认节点/关系类型**\n\n"
                )

            report_lines.append("\n")

    REPORT.parent.mkdir(parents=True, exist_ok=True)
    REPORT.write_text("".join(report_lines), encoding="utf-8")

    mapping_path = REPORT.parent / "tcm_dataset_field_mapping.md"
    mapping_path.write_text("".join(field_mapping_lines), encoding="utf-8")
    print(f"报告已生成: {REPORT}")
    print(f"字段映射已生成: {mapping_path}")

if __name__ == "__main__":
    main()

### 3.3 P1-0 Neo4jImporter 设计

```python
# backend/scripts/neo4j_importer/core.py
"""
通用 Neo4j 批量导入器

设计要点：
- UNWIND + MERGE 批量写入（避免单条语句往返）
- 幂等（重复执行不产生重复节点/关系）
- 自动建索引/约束
- 进度条 + 失败重试
"""
from dataclasses import dataclass, field
from pathlib import Path
from typing import Optional, Callable
import csv
import time

from langchain_community.graphs import Neo4jGraph


@dataclass
class NodeSpec:
    """节点导入规格"""
    label: str                              # 节点标签（如 "TCMSymptom"）
    csv_path: Path                          # CSV 文件路径
    id_field: str                           # CSV 中作为唯一 ID 的列名
    extra_fields: list[str] = field(default_factory=list)  # 额外导入的字段
    encoding: str = "utf-8"
    delimiter: str = "\t"


@dataclass
class RelSpec:
    """关系导入规格"""
    rel_type: str                           # 关系类型（如 "SYNDROME_ASSOCIATED_WITH_TCM_SYMPTOM"）
    csv_path: Path
    from_label: str                         # 起点节点标签
    from_id: str                            # CSV 中起点 ID 列
    to_label: str                           # 终点节点标签
    to_id: str                              # CSV 中终点 ID 列
    extra_fields: list[str] = field(default_factory=list)
    encoding: str = "utf-8"
    delimiter: str = "\t"


class Neo4jImporter:
    """通用 Neo4j 导入器（单例包装 graph）"""

    def __init__(self, graph: Neo4jGraph, batch_size: int = 1000):
        self.graph = graph
        self.batch_size = batch_size

    def create_constraints(self, specs: list[NodeSpec]):
        """为每个节点标签的唯一 ID 创建约束"""
        for spec in specs:
            cypher = f"""
            CREATE CONSTRAINT IF NOT EXISTS
            FOR (n:{spec.label}) REQUIRE n.{spec.id_field} IS UNIQUE
            """
            try:
                self.graph.query(cypher)
            except Exception as e:
                # 旧版 Neo4j 不支持 IF NOT EXISTS，容错
                print(f"⚠️ 约束 {spec.label}.{spec.id_field} 创建失败: {e}")

    def import_nodes(self, spec: NodeSpec) -> int:
        """批量导入节点"""
        all_props = []
        with open(spec.csv_path, encoding=spec.encoding) as f:
            reader = csv.DictReader(f, delimiter=spec.delimiter)
            for row in reader:
                props = {spec.id_field: row[spec.id_field]}
                for f_name in spec.extra_fields:
                    if row.get(f_name):
                        props[f_name] = row[f_name]
                all_props.append(props)

        # 批量写入
        total = 0
        cypher = f"""
        UNWIND $batch AS row
        MERGE (n:{spec.label} {{ {spec.id_field}: row.{spec.id_field} }})
        SET n += row
        """
        for i in range(0, len(all_props), self.batch_size):
            batch = all_props[i:i + self.batch_size]
            self.graph.query(cypher, params={"batch": batch})
            total += len(batch)
            print(f"  {spec.label}: {total}/{len(all_props)}")

        print(f"✅ {spec.label} 导入完成: {total} 条")
        return total

    def import_relationships(self, spec: RelSpec) -> int:
        """批量导入关系"""
        all_rows = []
        with open(spec.csv_path, encoding=spec.encoding) as f:
            reader = csv.DictReader(f, delimiter=spec.delimiter)
            for row in reader:
                rel_props = {
                    "from_id": row[spec.from_id],
                    "to_id": row[spec.to_id],
                }
                for f_name in spec.extra_fields:
                    if row.get(f_name):
                        rel_props[f_name] = row[f_name]
                all_rows.append(rel_props)

        # 批量 MATCH + MERGE
        total = 0
        cypher = f"""
        UNWIND $batch AS row
        MATCH (a:{spec.from_label} {{ {spec.from_id}: row.from_id }})
        MATCH (b:{spec.to_label} {{ {spec.to_id}: row.to_id }})
        MERGE (a)-[r:{spec.rel_type}]->(b)
        """
        if spec.extra_fields:
            set_clauses = ", ".join(
                [f"r.{f} = row.{f}" for f in spec.extra_fields]
            )
            cypher += f"\nSET {set_clauses}"

        for i in range(0, len(all_rows), self.batch_size):
            batch = all_rows[i:i + self.batch_size]
            try:
                self.graph.query(cypher, params={"batch": batch})
                total += len(batch)
                print(f"  {spec.rel_type}: {total}/{len(all_rows)}")
            except Exception as e:
                print(f"❌ {spec.rel_type} 批次 {i} 失败: {e}")
                # 可选：降级为单条写入重试
                for row in batch:
                    try:
                        self.graph.query(cypher, params={"batch": [row]})
                        total += 1
                    except Exception:
                        pass

        print(f"✅ {spec.rel_type} 导入完成: {total} 条")
        return total
```

async def _query_similar_syndromes(
    symptoms: list[str], top_k: int = 5
) -> list[dict]:
    """
    从 Neo4j 查询与症状最相似的证型

    新 Schema：TCMSymptom 与 Syndrome 通过 SYNDROME_ASSOCIATED_WITH_TCM_SYMPTOM 关联
    """
    graph = get_neo4j_graph(database="tcm_graph")
    if graph is None:
        logger.warning("Neo4j 不可用，跳过证型查询")
        return []

    try:
        result = graph.query(
            """
            MATCH (s:TCMSymptom)-[r:SYNDROME_ASSOCIATED_WITH_TCM_SYMPTOM]->(sy:Syndrome)
            WHERE s.name IN $symptoms
            RETURN sy.name AS syndrome,
                   count(r) AS match_count,
                   collect(s.name) AS matched_symptoms,
                   avg(coalesce(r.weight, 1.0)) AS avg_weight
            ORDER BY match_count DESC, avg_weight DESC
            LIMIT $top_k
            """,
            params={"symptoms": symptoms, "top_k": top_k},
        )
        return [dict(record) for record in result]
    except Exception as e:
        logger.error(f"证型查询失败: {e}")

### 5.1 Task 1：`_query_similar_cases` 接 RAG

#### 5.1.1 目标

`_query_similar_cases()` 当前返回 mock，改为基于**真实 Neo4j 数据**做相似医案检索。

#### 5.1.2 设计选择（两种方案）

| 方案 | 优势 | 劣势 | 推荐 |
|------|------|------|------|
| **A. Neo4j 子图匹配** | 不需要额外的向量库；直接利用已灌库的图 | 相似度语义弱（仅按症状重合度排序） | ✅ **首选** |
| B. 向量检索（FAISS / pgvector） | 语义级相似度 | 需要额外基础设施 + embedding 成本 | 备选 |

**推荐方案 A**：基于 Neo4j 中已有的 `TCMSymptom` / `Disease` / `Syndrome` 节点，找"症状最相似的证型 + 对应方剂"，作为相似医案返回。

#### 5.1.3 实现

**位置**：`backend/app/src/agent/components/diagnose/nodes/moderate_diagnosis/moderate_diagnosis.py` 中的 `_query_similar_cases()`

```python
async def _query_similar_cases(
    symptoms: list[str], top_k: int = 3
) -> list[dict]:
    """
    基于 Neo4j 查询相似医案（症状 → 证型 → 方剂）

    路径：
    用户的症状列表
      → 匹配 TCMSymptom 节点
      → 通过 SYNDROME_ASSOCIATED_WITH_TCM_SYMPTOM 找到对应证型
      → 通过 ITCM 的 FORMULA_CONTAINS_HERB 找到方剂
    """
    graph = get_neo4j_graph(database="tcm_graph")
    if graph is None:
        logger.warning("Neo4j 不可用，跳过相似医案查询")
        return []

    try:
        result = graph.query(
            """
            MATCH (s:TCMSymptom)-[:SYNDROME_ASSOCIATED_WITH_TCM_SYMPTOM]->(sy:Syndrome)
            WHERE s.name IN $symptoms
            WITH sy, count(s) AS match_count
            ORDER BY match_count DESC
            LIMIT $top_k

            MATCH (sy)<-[:FORMULA_CONTAINS_HERB*0..1]-(p:Formula)
            OPTIONAL MATCH (sy)-[:SYNDROME_ASSOCIATED_WITH_TCM_SYMPTOM]->(matched:TCMSymptom)
            RETURN sy.name AS syndrome,
                   sy.description AS syndrome_description,
                   p.name AS similar_formula,
                   p.composition AS composition,
                   match_count AS symptom_match_count,
                   collect(DISTINCT matched.name) AS evidence_symptoms
            ORDER BY match_count DESC
            LIMIT $top_k
            """,
            params={"symptoms": symptoms, "top_k": top_k},
        )
        return [dict(record) for record in result]
    except Exception as e:
        logger.error(f"相似医案查询失败: {e}")
        return []
```

from pathlib import Path

def export_herb_corpus(graph):
    """导出 Herb 节点为 markdown"""
    herbs = graph.query("""
        MATCH (h:Herb)
        OPTIONAL MATCH (h)-[:HERB_CONTAINS_INGREDIENT]->(i:Ingredient)
        WITH h, collect(DISTINCT i.name) AS ingredients
        RETURN h.chinese_name AS name, h.nature AS nature, h.taste AS taste,
               h.meridian AS meridian, h.effect AS effect, ingredients
    """)
    out_file = OUTPUT_DIR / "herbs.md"
    with open(out_file, "w", encoding="utf-8") as f:
        for h in herbs:
            f.write(f"# {h['name']}\n\n")
            f.write(f"- **性味**: {h.get('nature', '')} {h.get('taste', '')}\n")
            f.write(f"- **归经**: {h.get('meridian', '')}\n")
            f.write(f"- **功效**: {h.get('effect', '')}\n")
            if h.get('ingredients'):
                f.write(f"- **主要成分**: {', '.join(h['ingredients'])}\n")
            f.write("\n---\n\n")
    print(f"✅ herbs.md: {len(herbs)} 条")

def export_formula_corpus(graph):
    """导出 Formula 节点为 markdown"""
    formulas = graph.query("""
        MATCH (p:Formula)

        for p in formulas:

            f.write(f"- **出处**: {p.get('source', '')}\n")

                f.write(f"- **药材**: {', '.join(p['herbs'])}\n")

## 6. 风险与约束

### 6.1 数据规模风险

- TCMBank + SymMap_v2 + ITCM 合并后预计实体数 **万级**，关系数 **十万级**
- Neo4j 默认配置可能不够，需评估 `neo4j.conf` 的 `dbms.memory.heap.max_size`
- 导入时间可能较长（小时级），建议先做小批量试点

### 6.2 编码与目录风险

- 多个目录含空格（`TCM symptom/`、`MM symptom/`、`Network used in TOOL/`），脚本需用 `Path` 而非字符串拼接
- 文件可能是 GBK 编码（特别是较老的数据库），导入器需编码兜底

### 6.3 跨库节点冲突

- TCMBank / SymMap_v2 / ITCM 都有 Herb 节点，**ID 体系不同**
- 方案：每个数据库导入时用带前缀的 ID（如 `TCMBANK_H001`、`SYMMAP_H001`、`ITCM_H001`），避免冲突
- 跨库关系用 `MATCH` 前缀判断

### 6.4 P3 依赖风险

- Task 1（`_query_similar_cases` RAG）和 Task 2（origin_data 清理）**强依赖 P1+P2 灌库成功**
- 如果 P1+P2 因 Neo4j 资源不足或数据质量问题失败，P3 需重新评估方案
